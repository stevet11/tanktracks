cmake_minimum_required(VERSION 3.14)
project(tanktracks C)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# raylib: use the vendored copy if present, otherwise download it.
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/raylib-5.5/CMakeLists.txt)
  add_subdirectory(third_party/raylib-5.5 raylib EXCLUDE_FROM_ALL)
else()
  include(FetchContent)
  FetchContent_Declare(raylib URL https://github.com/raysan5/raylib/archive/refs/tags/5.5.tar.gz)
  FetchContent_MakeAvailable(raylib)
endif()

# ENet (networking): use the vendored copy if present, otherwise download it. Built straight from its sources.
if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/third_party/enet-1.3.18/host.c)
  set(ENET_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/enet-1.3.18)
else()
  include(FetchContent)
  FetchContent_Declare(enetsrc URL https://github.com/lsalzman/enet/archive/refs/tags/v1.3.18.tar.gz SOURCE_SUBDIR _none_)
  FetchContent_MakeAvailable(enetsrc)
  set(ENET_DIR ${enetsrc_SOURCE_DIR})
endif()
set(ENET_SRC callbacks compress host list packet peer protocol)
if(WIN32)
  list(APPEND ENET_SRC win32)
else()
  list(APPEND ENET_SRC unix)
endif()
list(TRANSFORM ENET_SRC PREPEND ${ENET_DIR}/)
list(TRANSFORM ENET_SRC APPEND .c)
add_library(enet STATIC ${ENET_SRC})
target_include_directories(enet PUBLIC ${ENET_DIR}/include)
if(NOT WIN32)
  target_compile_definitions(enet PRIVATE HAS_FCNTL=1 HAS_POLL=1 HAS_GETADDRINFO=1 HAS_GETNAMEINFO=1 HAS_GETHOSTBYNAME_R=1 HAS_GETHOSTBYADDR_R=1 HAS_INET_PTON=1 HAS_INET_NTOP=1 HAS_MSGHDR_FLAGS=1 HAS_SOCKLEN_T=1)
else()
  target_link_libraries(enet PUBLIC ws2_32 winmm)
endif()

add_executable(tanktracks main.c net.c)
target_link_libraries(tanktracks PRIVATE raylib enet)
if(UNIX AND NOT APPLE)
  target_link_libraries(tanktracks PRIVATE m)
endif()
if(MSVC)
  target_compile_definitions(tanktracks PRIVATE _CRT_SECURE_NO_WARNINGS)
  set_target_properties(tanktracks PROPERTIES LINK_FLAGS "/SUBSYSTEM:CONSOLE")
endif()
